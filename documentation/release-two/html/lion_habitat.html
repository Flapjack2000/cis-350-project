207    def _draw_lion(self, animal: Animal, screen: pygame.Surface) -> None:
208        """Draw the lion on the screen.
209
210        Args:
211            animal (Animal): the lion
212            screen (pygame.Surface): the screen to draw on
213        """
214        should_flip = animal.facing_left != animal.default_facing_left
215        body_w = animal.layers[7].get_width() if animal.layers else 0
216        body_pos = pygame.Vector2(animal.x + body_w / 2, animal.y)
217
218        pivots = []
219        for i, layer in enumerate(animal.layers):
220            name = animal.layer_files[i]
221            if "hind_back_upper" in name or "hind_front_upper" in name:
222                pivots.append(self._HIND_PIVOT)
223            elif "fore_back_upper" in name or "fore_front_upper" in name:
224                pivots.append(self._FRONT_PIVOT)
225            else:
226                pivots.append(
227                    pygame.Vector2(
228                        layer.get_width() // 2,
229                        layer.get_height() // 2
230                    )
231                )
232
233        for layer, angle, pivot in (
234                zip(animal.layers, animal.layer_angles, pivots)):
235            img, rect = MathHelper.rotate_image(layer, angle, pivot)
236            rect.center = int(body_pos[0]), int(body_pos[1])
237            if should_flip:
238                img = pygame.transform.flip(img, True, False)
239            screen.blit(img, rect)

286                    prev_mode = self._water_active or self._feed_active

290                    if new_mode != prev_mode:

292
293                if self._water_active and w_rect.collidepoint(e.pos):
294                    self._water_level = min(100, self._water_level + 2)
295                elif self._feed_active and f_rect.collidepoint(e.pos):
296                    self._feed_level = min(100, self._feed_level + 2)
297
298                for i, p in enumerate(self._waste_positions):
299                    if not self._waste_clicked[i]:
300                        r = self._waste_sprite.get_rect(
301                            center=(int(p.x), int(p.y))

306    def update(self, dt: float) -> None:
307        """Update the game state.
308
309        Args:
310            dt (float): the time since the last frame
311        """
312
313        if self._interaction_timer > 0:
314            self._interaction_timer -= dt
315            if self._interaction_timer <= 0 and self._animals:
316                self._animals[0].speed = self._SPEED
317            super().update(dt)
318            return
319
320        super().update(dt)
321        if not self._animals or not (self._water_active or self._feed_active):
322            return
323
324        lion = self._animals[0]
325        w_rect, f_rect = self._get_station_rects()
326        body_w = lion.layers[7].get_width() if lion.layers else 0
327        mx = lion.x + body_w // 2
328
329        current_zone = None
330        if self._water_active and w_rect.left <= mx <= w_rect.right:
331            current_zone = "water"
332        elif self._feed_active and f_rect.left <= mx <= f_rect.right:
333            current_zone = "feed"
334
335        if current_zone:
336            if not self._pass_counted:
337                self._pass_counted = True
338                if current_zone == "water":
339                    self._water_passes += 1
340                    should_stop = self._water_passes >= self._water_threshold
341                else:
342                    self._feed_passes += 1
343                    should_stop = self._feed_passes >= self._feed_threshold
344
345                level = self._water_level if (
346                        current_zone == "water") else self._feed_level
347
348                if should_stop and level > 0:
349                    lion.speed = 0
350                    self._interaction_timer = 1.5
351                    if current_zone == "water":
352                        self._water_level = max(0, self._water_level - 40)
353                        self._water_passes = 0
354                        self._water_threshold = random.randint(1, 3)
355                    else:
356                        self._feed_level = max(0, self._feed_level - 40)
357                        self._feed_passes = 0
358                        self._feed_threshold = random.randint(1, 3)
359                else:
360                    lion.speed = self._SPEED
361        else:
362            self._pass_counted = False

426            self,
427            screen: pygame.Surface,
428            rect: pygame.Rect,
429            level: float | int,
430            fill_col: tuple[int, int, int],
431            border_col: tuple[int, int, int]
432    ) -> None:
433        """Draw a station.
434
435        Args:
436            screen (pygame.Surface): the screen to draw on
437            rect (pygame.Rect): the rect of the station
438            level (float | int): the station contents level
439            fill_col (tuple[int, int, int]): the station fill color
440            border_col (tuple[int, int, int]): the station border color
441        """
442
443        s1, s2 = self._STATION_S1, self._STATION_S2
444        fill_h = int((s1 - s2) * (level / 100))
445        pygame.draw.rect(
446            screen,
447            fill_col,
448            (rect.x + s2, rect.y + s1 - s2 - fill_h, s1 - s2 * 2, fill_h)
449        )
450        pygame.draw.rect(screen, border_col,
451                         (rect.x, rect.y + s1 - s2, s1, s2))
452        pygame.draw.rect(screen, border_col,
453                         (rect.x, rect.y, s2, s1))
454        pygame.draw.rect(screen, border_col,
455                         (rect.x + s1 - s2, rect.y, s2, s1))
