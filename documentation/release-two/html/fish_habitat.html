 21class _Fish:
 22    """Represents a single fish in the aquarium habitat."""
 23
 24    def __init__(
 25            self, img_path: str, faces_right: bool, size_px: int,
 26            x: float, y: float, speed: float, screen_width: int
 27    ) -> None:
 28        raw = pygame.image.load(img_path).convert_alpha()
 29        w, h = raw.get_size()
 30        scale = size_px / h
 31        self.base_img = (
 32            pygame.transform.smoothscale(
 33                raw,
 34                (int(w * scale),
 35                 size_px
 36                 )
 37            )
 38        )
 39
 40        self.faces_right = faces_right
 41        self.x = x
 42        self.y = y
 43        self.home_y = y
 44        self.speed = speed
 45        self.screen_width = screen_width
 46        self.moving_right = random.choice([True, False])
 47
 48    @property
 49    def width(self) -> int:
 50        """Return the width of the fish's image."""
 51        return self.base_img.get_width()
 52
 53    def _respawn(self) -> None:
 54        """Respawn fish."""
 55        self.moving_right = random.choice([True, False])
 56        self.x = -self.width if self.moving_right else float(self.screen_width)
 57        self.y = self.home_y
 58
 59    def update(self, dt: float) -> None:
 60        """Update the position of the fish.
 61
 62        Args:
 63            dt (float): time since last frame
 64        """
 65        move = self.speed * dt * 60
 66        if self.moving_right:
 67            self.x += move
 68            if self.x >= self.screen_width:
 69                self._respawn()
 70        else:
 71            self.x -= move
 72            if self.x + self.width <= 0:
 73                self._respawn()
 74
 75    def draw(self, surface: pygame.Surface) -> None:
 76        """Render fish on the screen.
 77
 78        Args:
 79            surface (pygame.Surface): the surface to draw on
 80        """
 81        should_flip = self.moving_right != self.faces_right
 82        img = pygame.transform.flip(self.base_img, should_flip, False)
 83        surface.blit(img, (int(self.x), int(self.y)))

189    def handle_events(self, events: list[pygame.event.Event]) -> None:
190        """Handle mouse events and pausing.
191
192        Args:
193            events (list[pygame.event.Event]):
194                the events to handle
195        """
196        mouse_pos = pygame.mouse.get_pos()
197
198        for event in events:
199            if event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE:
200                from pause_scene import PauseScene
201                self._manager.push(PauseScene(self._manager))
202
203            if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
204                if (
205                        self._feed_task_active
206                        and self.feed_icon_rect.collidepoint(mouse_pos)
207                ):
208                    self.feeding = True
209
210                if self._btn_checklist.rect.collidepoint(mouse_pos):
211                    from checklist_scene import ChecklistScene
212                    self._manager.push(
213                        ChecklistScene(
214                            self._manager,
215                            self._manager.context.checklist
216                        )
217                    )
218
219                if self._btn_map.rect.collidepoint(mouse_pos):
220                    from map_scene import MapScene
221                    self._manager.push(MapScene(self._manager))
222
223            if event.type == pygame.MOUSEBUTTONUP and event.button == 1:
224                self.feeding = False
225
226    def update(self, dt: float) -> None:
227        """Update fish positions and game state.
228
229        Args:
230            dt (float): time since last frame
231        """
232
233        # Update fish positions
234        for fish in self._fish:
235            fish.update(dt)
236
237        # Handle feeding updates
238        if self.feeding:
239            if pygame.mouse.get_pressed()[0]:
240                self.feed_progress += self.FEED_RATE * dt
241            else:
242                self.feed_progress -= self.DECAY_RATE * dt
243
244            self.feed_progress = max(0.0, min(100.0, self.feed_progress))
245
246            if self.feed_progress >= 100.0:
247                self._handle_feed_complete()
248
249        elif self.feed_progress > 0:
250            self.feed_progress -= self.DECAY_RATE * dt
251            self.feed_progress = max(0.0, self.feed_progress)
252
253    def _handle_feed_complete(self) -> None:
254        """Handle completion of feeding game."""
255
256        # Update game state
257        self.feed_progress = 100.0
258        self.feeding = False
259        self._feed_task_active = False
260
261        # Mark task complete
262        self._manager.context.checklist.complete_task("fish_feed")
263
264        # Exit to map and superimpose the checklist
265        from checklist_scene import ChecklistScene
266        self._manager.pop()
267        self._manager.push(
268            ChecklistScene(
269                self._manager,
270                self._manager.context.checklist
271            )
272        )

274    def draw(self, screen: pygame.Surface) -> None:
275        """
276         Args:
277            screen (pygame.Surface): the screen to draw on
278        """
279
280        # Draw background
281        if self._background:
282            screen.blit(self._background, (0, 0))
283        else:
284            screen.fill((30, 100, 180))
285
286        # Render fish
287        for fish in self._fish:
288            fish.draw(screen)
289
290        # Draw Interface
291        icon = (
292            self.feed_icon if self._feed_task_active else self.feed_icon_grey
293        )
294        screen.blit(icon, self.feed_icon_rect)
295        self._btn_checklist.draw(screen)
296        self._btn_map.draw(screen)
297
298        # Drag food can
299        if self.feeding:
300            mouse_pos = pygame.mouse.get_pos()
301            cursor_rect = self.feed_bottle.get_rect(center=mouse_pos)
302            screen.blit(self.feed_bottle, cursor_rect)
303            self._draw_progress_bar(screen)

305    def _draw_progress_bar(self, screen: pygame.Surface) -> None:
306        """Render the food game progress bar.
307
308        Args:
309            screen (pygame.Surface): the screen to draw on
310        """
311        bar_w, bar_h = 220, 18
312        screen_w = screen.get_width()
313        bar_x = (screen_w - bar_w) // 2

316        pygame.draw.rect(
317            screen,
318            (60, 60, 60),
319            (bar_x, bar_y, bar_w, bar_h),
320            border_radius=4
321        )
322        fill_w = int(bar_w * (self.feed_progress / 100))
323        pygame.draw.rect(
324            screen,
325            (80, 220, 120),
326            (bar_x, bar_y, fill_w, bar_h),
327            border_radius=4
328        )
