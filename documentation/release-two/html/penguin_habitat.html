255    def update(self, dt: float) -> None:
256        """Update game state.
257
258        Args:
259            dt (float): the time since the last frame
260        """
261        if self._interaction_timer > 0:
262            self._interaction_timer -= dt
263            if self._interaction_timer <= 0 and self._animals:
264                self._animals[0].speed = 0.5
265            super().update(dt)
266            return
267
268        super().update(dt)
269
270        if not self._animals or not (self._water_active or self._feed_active):
271            return
272
273        penguin = self._animals[0]
274        w_rect, f_rect = self._get_station_rects()
275        px = penguin.x + (penguin.layers[0].get_width() //
276                          2 if penguin.layers else 0)
277
278        current_zone = None
279        if self._water_active and w_rect.left <= px <= w_rect.right:
280            current_zone = "water"
281        elif self._feed_active and f_rect.left <= px <= f_rect.right:
282            current_zone = "feed"
283
284        if current_zone:
285            if not self._pass_counted:
286                self._pass_counted = True
287                if current_zone == "water":
288                    self._water_passes += 1
289                    should_stop = self._water_passes >= self._water_threshold
290                else:
291                    self._feed_passes += 1
292                    should_stop = self._feed_passes >= self._feed_threshold
293
294                level = self._water_level if (
295                        current_zone == "water") else self._feed_level
296
297                if should_stop and level > 0:
298                    penguin.speed = 0
299                    self._interaction_timer = 2.0
300                    if current_zone == "water":
301                        self._water_level = max(0, self._water_level - 35)
302                        self._water_passes = 0
303                        self._water_threshold = random.randint(1, 3)
304                    else:
305                        self._feed_level = max(0, self._feed_level - 35)
306                        self._feed_passes = 0
307                        self._feed_threshold = random.randint(1, 3)
308                else:
309                    penguin.speed = 0.5
310        else:
311            self._pass_counted = False
312
313        if self._water_active and self._water_level >= 100:
314            self._water_active = False
315            self._complete_task("penguin_water")
316
317        if self._feed_active and self._feed_level >= 100:
318            self._feed_active = False
319            self._complete_task("penguin_feed")

361                    lines.append(line)

363            lines.append(line)
364
365            y_off = w_rect.top + 20
366            for ln in lines:
367                surf = self._font.render(ln.strip(), True, (0, 0, 0))
368                screen.blit(surf, (40, y_off))

371    def _draw_station(
372            self,
373            screen: pygame.Surface,
374            rect: pygame.Rect,
375            level: float | int,
376            fill_col: tuple[int, int, int],
377            border_col: tuple[int, int, int]
378    ) -> None:
379        """Draw the station on the screen.
380
381        Args:
382            screen (pygame.Surface): the screen to draw on
383            rect (pygame.Rect): the rect of the station
384            level (float | int): the level of the station's contents
385            fill_col (tuple[int, int, int]): the fill color of the station
386            border_col (tuple[int, int, int]): the border color of the station
387        """
388        s1, s2 = self._STATION_S1, self._STATION_S2
389        fill_h = int((s1 - s2) * (level / 100))
390        (pygame.draw.rect
391         (screen, fill_col, (rect.x + s2, rect.y + s1 - s2 - fill_h,
392                             s1 - s2 * 2, fill_h)))
393        (pygame.draw.rect
394         (screen, border_col, (rect.x, rect.y + s1 - s2, s1, s2)))
395        (pygame.draw.rect
396         (screen, border_col, (rect.x, rect.y, s2, s1)))
397        (pygame.draw.rect
398         (screen, border_col, (rect.x + s1 - s2, rect.y, s2, s1)))
