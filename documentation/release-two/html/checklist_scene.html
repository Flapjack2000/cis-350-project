108    def handle_events(self, events: list[pygame.event.Event]) -> None:
109        """Handle mouse events and pausing.
110
111        Args:
112            events (list[pygame.event.Event]): the events to handle
113        """
114
115        mouse_pos = pygame.mouse.get_pos()
116        for event in events:
117            if event.type == pygame.KEYDOWN and event.key == pygame.K_ESCAPE:
118                from pause_scene import PauseScene
119                self._manager.push(PauseScene(self._manager))
120
121            if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
122                for name, button in self.buttons.items():
123                    if button.is_clicked(mouse_pos, (True,)):
124                        self._handle_action(name)
125

127        """Handle return button click.
128
129        Args:
130            action (str): the action to take
131        """
132        if action == "return":
133            self._manager.pop()
